19const LABEL: Style = Style::new().fg(Color::DarkGray);
20const ACCENT: Style = Style::new().fg(Color::Cyan);
21const MUTED: Style = Style::new().fg(Color::Gray);
22const HIGHLIGHT: Style = Style::new().fg(Color::Yellow).add_modifier(Modifier::BOLD);
23const ACTIVE: Style = Style::new().fg(Color::Green);

25/// Draws the TUI chrome and active pane.
26pub fn draw(frame: &mut Frame<'_>, model: &AppModel) {
27    let chunks = Layout::default()
28        .direction(Direction::Vertical)
29        .constraints([
30            Constraint::Length(3),
31            Constraint::Min(8),
32            Constraint::Length(2),
33        ])
34        .split(frame.area());
35
36    draw_title(frame, chunks[0], model);
37    match model.view {
38        ViewMode::Results => draw_results(frame, chunks[1], model),
39        ViewMode::Builder => draw_builder(frame, chunks[1], model),
40        ViewMode::Payload => draw_payload(frame, chunks[1], model),
41        ViewMode::Codegen => draw_codegen(frame, chunks[1], model),
42        ViewMode::Help => draw_help(frame, chunks[1]),
43    }
44    draw_footer(frame, chunks[2], model);
45}

47fn draw_title(frame: &mut Frame<'_>, area: Rect, model: &AppModel) {
48    let preset = model.config.preset.as_deref().unwrap_or("(defaults)");
49    let tabs = [
50        ViewMode::Results.tab_label(),
51        ViewMode::Builder.tab_label(),
52        ViewMode::Payload.tab_label(),
53        ViewMode::Codegen.tab_label(),
54    ];
55    let tab_spans: Vec<Span<'_>> = tabs
56        .iter()
57        .flat_map(|label| {
58            let style = if *label == model.view.tab_label() {
59                ACCENT.add_modifier(Modifier::BOLD)
60            } else {
61                MUTED
62            };
63            [Span::styled(*label, style), Span::raw("  ")]
64        })
65        .collect();
66    let title = Paragraph::new(Line::from(vec![
67        Span::styled("tvscreener-tui", ACCENT.add_modifier(Modifier::BOLD)),
68        Span::raw("  "),
69        Span::styled(preset, MUTED),
70        Span::raw("  "),
71    ]))
72    .block(
73        Block::default()
74            .borders(Borders::ALL)
75            .title(Line::from(tab_spans)),
76    );
77    frame.render_widget(title, area);
78}

80fn draw_footer(frame: &mut Frame<'_>, area: Rect, model: &AppModel) {
81    let status = model.error.as_deref().unwrap_or(model.status.as_str());
82    let style = if model.error.is_some() {
83        Style::new().fg(Color::Red)
84    } else {
85        MUTED
86    };
87    let keys = match model.view {
88        ViewMode::Results => {
89            "Tab views · r refresh · a watch · c copy rows · h help · ↑↓ scroll"
90        }
91        ViewMode::Builder => {
92            "Tab views · ←→ asset/preset · +/- limit · s search · t/u sort · m/i scope · f filter · Enter scan · j/k focus"
93        }
94        ViewMode::Payload | ViewMode::Codegen => "Tab views · c copy · ↑↓ scroll · h help",
95        ViewMode::Help => "Tab views · h back · q quit",
96    };
97    let line = Paragraph::new(Line::from(vec![
98        Span::styled(status, style),
99        Span::raw("  "),
100        Span::styled(keys, LABEL),
101    ]));
102    frame.render_widget(line, area);
103}

140fn draw_builder(frame: &mut Frame<'_>, area: Rect, model: &AppModel) {
141    let mut lines = builder_header_lines(model);
142    lines.extend(builder_filter_lines(model));
143    lines.push(Line::from(""));
144    lines.extend(input_hint_lines(model));
145
146    let body = Paragraph::new(lines)
147        .wrap(Wrap { trim: false })
148        .scroll((u16::try_from(model.scroll).unwrap_or(0), 0))
149        .block(Block::default().borders(Borders::ALL).title(" Builder "));
150    frame.render_widget(body, area);
151}

153fn builder_header_lines(model: &AppModel) -> Vec<Line<'static>> {
154    let preset_name = model
155        .builder
156        .preset_options
157        .get(model.builder.preset_index)
158        .map_or("(defaults)", String::as_str);
159    let search = optional_config_text(model.config.search.as_deref());
160    let sort = optional_config_text(model.config.sort_by.as_deref());
161    let sort_dir = if model.config.ascending {
162        "asc"
163    } else {
164        "desc"
165    };
166    let markets = stock_scope_text(model, model.config.markets.as_deref());
167    let index = stock_scope_text(model, model.config.index.as_deref());
168    vec![
169        Line::from(Span::styled(
170            "Query builder",
171            ACCENT.add_modifier(Modifier::BOLD),
172        )),
173        Line::from(""),
174        builder_row(
175            model,
176            BuilderFocus::Asset,
177            "Asset",
178            &format!("{}  (← → cycle)", model.config.asset.as_str()),
179        ),
180        builder_row(
181            model,
182            BuilderFocus::Preset,
183            "Preset",
184            &format!("{preset_name}  (← → cycle)"),
185        ),
186        builder_row(
187            model,
188            BuilderFocus::Limit,
189            "Limit",
190            &format!("{} rows  (+ / -)", model.config.limit),
191        ),
192        builder_row(
193            model,
194            BuilderFocus::Search,
195            "Search",
196            &format!("{search}  (s edit)"),
197        ),
198        builder_row(
199            model,
200            BuilderFocus::Sort,
201            "Sort",
202            &format!("{sort} {sort_dir}  (t edit · u toggle)"),
203        ),
204        builder_row(
205            model,
206            BuilderFocus::Markets,
207            "Markets",
208            &format!("{markets}  (m edit)"),
209        ),
210        builder_row(
211            model,
212            BuilderFocus::Index,
213            "Index",
214            &format!("{index}  (i edit)"),
215        ),
216        Line::from(""),
217        builder_row(
218            model,
219            BuilderFocus::Filters,
220            "Filters",
221            "(f add · d remove · j/k select)",
222        ),
223    ]
224}

226fn builder_filter_lines(model: &AppModel) -> Vec<Line<'static>> {
227    if model.config.filters.is_empty() {
228        return vec![Line::from(Span::styled("  (no filters)", MUTED))];
229    }
230    model
231        .config
232        .filters
233        .iter()
234        .enumerate()
235        .map(|(i, f)| {
236            let marker = if i == model.builder.filter_selected {
237                ">"
238            } else {
239                " "
240            };
241            let style = if i == model.builder.filter_selected {
242                HIGHLIGHT
243            } else {
244                MUTED
245            };
246            Line::from(Span::styled(
247                format!("  {marker} {} {} {}", f.left, f.operation, f.value),
248                style,
249            ))
250        })
251        .collect()
252}

254fn stock_scope_text(model: &AppModel, value: Option<&str>) -> String {
255    if model.config.asset != crate::field::Asset::Stock {
256        return "(n/a)".into();
257    }
258    value
259        .filter(|s| !s.is_empty())
260        .map_or_else(|| "(none)".into(), String::from)
261}
262
263fn optional_config_text(value: Option<&str>) -> String {
264    value
265        .filter(|s| !s.is_empty())
266        .map_or_else(|| "(none)".into(), String::from)
267}

269fn builder_row(model: &AppModel, row: BuilderFocus, label: &str, value: &str) -> Line<'static> {
270    let style = if model.builder.focus == row {
271        HIGHLIGHT
272    } else {
273        Style::new()
274    };
275    Line::from(vec![
276        Span::styled(format!("{label:8}"), style),
277        Span::raw(" "),
278        Span::styled(
279            value.to_string(),
280            if model.builder.focus == row {
281                ACTIVE
282            } else {
283                MUTED
284            },
285        ),
286    ])
287}

289fn prompt_input_lines(label: &str, buf: &str) -> Vec<Line<'static>> {
290    vec![
291        Line::from(""),
292        Line::from(Span::styled(format!("{label}: {buf}_"), ACTIVE)),
293        Line::from(Span::styled("Enter save · Esc cancel", LABEL)),
294    ]

297fn input_hint_lines(model: &AppModel) -> Vec<Line<'static>> {
298    match model.builder.input_mode {
299        InputMode::None => Vec::new(),
300        InputMode::Search => prompt_input_lines("Search", &model.builder.input_buf),
301        InputMode::Sort => prompt_input_lines("Sort field", &model.builder.input_buf),
302        InputMode::Markets => prompt_input_lines("Markets CSV", &model.builder.input_buf),
303        InputMode::Index => prompt_input_lines("Index CSV", &model.builder.input_buf),
304        InputMode::FilterField => {
305            let mut out = vec![
306                Line::from(""),
307                Line::from(Span::styled(
308                    format!("Field search: {}_", model.builder.input_buf),
309                    ACTIVE,
310                )),
311            ];
312            for (i, field) in model.builder.field_matches.iter().take(8).enumerate() {
313                let pick = if i == model.builder.field_pick {
314                    ">"
315                } else {
316                    " "
317                };
318                let style = if i == model.builder.field_pick {
319                    HIGHLIGHT
320                } else {
321                    MUTED
322                };
323                out.push(Line::from(Span::styled(
324                    format!("  {pick} {} ({})", field.field_name, field.label),
325                    style,
326                )));
327            }
328            out.push(Line::from(Span::styled(
329                "j/k pick · Enter next · Esc cancel",
330                LABEL,
331            )));
332            out
333        }
334        InputMode::FilterOp => {
335            let op = FILTER_OPS
336                .get(model.builder.op_index)
337                .copied()
338                .unwrap_or(FILTER_OPS[0]);
339            vec![
340                Line::from(""),
341                Line::from(Span::styled(
342                    format!("Operator: {op}  (o cycle · Enter value)"),
343                    ACTIVE,
344                )),
345            ]
346        }
347        InputMode::FilterValue => vec![
348            Line::from(""),
349            Line::from(Span::styled(
350                format!(
351                    "Value for {}: {}_",
352                    model.builder.draft_field.as_deref().unwrap_or("?"),
353                    model.builder.input_buf
354                ),
355                ACTIVE,
356            )),
357            Line::from(Span::styled("Enter add · Esc cancel", LABEL)),
358        ],
359    }
360}

362fn draw_payload(frame: &mut Frame<'_>, area: Rect, model: &AppModel) {
363    let text = match payload_pretty(&model.config, false) {
364        Ok(json) => json,

366    };
367    let body = Paragraph::new(text)
368        .wrap(Wrap { trim: false })
369        .scroll((u16::try_from(model.scroll).unwrap_or(0), 0))
370        .style(MUTED)
371        .block(
372            Block::default()
373                .borders(Borders::ALL)
374                .title(" Payload (build_payload JSON) "),
375        );
376    frame.render_widget(body, area);
377}

393fn draw_results(frame: &mut Frame<'_>, area: Rect, model: &AppModel) {
394    let selected = select_table_fields(&model.rows, &model.fields, model.max_columns);
395    let columns = get_columns_to_request(&model.fields);
396
397    let mut header_cells = vec![Cell::from("Symbol").style(ACCENT.add_modifier(Modifier::BOLD))];
398    header_cells.extend(
399        selected
400            .iter()
401            .map(|f| Cell::from(f.label.as_str()).style(ACCENT.add_modifier(Modifier::BOLD))),
402    );
403    let header = Row::new(header_cells).height(1);
404
405    let visible = model.rows.get(model.scroll..).unwrap_or(&[]);
406    let body_rows: Vec<Row<'_>> = visible
407        .iter()
408        .map(|row| {
409            let tech = RowTechMap::from_row(row, &columns);
410            let mut cells = vec![Cell::from(row.symbol.as_str())];
411            for field in &selected {
412                let value = row.data.get(&field.label).unwrap_or(&Value::Null);
413                let cell = format_cell_for_field(value, field, Some(&tech));
414                let style = Style::new().fg(tone_color(cell.tone));
415                cells.push(Cell::from(cell.text).style(style));
416            }
417            Row::new(cells)
418        })
419        .collect();
420
421    let mut widths = vec![Constraint::Length(22)];
422    widths.extend(selected.iter().map(|_| Constraint::Min(10)));
423
424    let table = Table::new(body_rows, widths)
425        .header(header)
426        .block(Block::default().borders(Borders::ALL).title(" Results "))
427        .column_spacing(1);
428
429    frame.render_widget(table, area);
430}

476    fn buffer_text(terminal: &Terminal<TestBackend>) -> String {
477        let buf = terminal.backend().buffer();
478        let area = buf.area();
479        let mut out = String::new();
480        for y in 0..area.height {
481            for x in 0..area.width {
482                out.push_str(buf[(x, y)].symbol());
483            }
484            out.push('\n');
485        }
486        out
487    }
