4//! `tvscreener-tui` - Ratatui screener TUI.
5//!
6//! ```bash
7//! cargo run --bin tvscreener-tui -- crypto --limit 10
8//! make run-tui ARGS='crypto --preset crypto_price --limit 5'
9//! make run-tui ARGS='crypto --watch --interval 30'
10//! ```

31#[derive(Debug, Clone, Copy, ValueEnum)]
32enum AssetArg {
33    Stock,
34    Crypto,
35    Forex,
36    Bond,
37    Futures,
38    Coin,
39}
40
41impl AssetArg {
42    const fn asset(self) -> Asset {
43        match self {
44            Self::Stock => Asset::Stock,
45            Self::Crypto => Asset::Crypto,
46            Self::Forex => Asset::Forex,
47            Self::Bond => Asset::Bond,
48            Self::Futures => Asset::Futures,
49            Self::Coin => Asset::Coin,
50        }
51    }
52}

54#[derive(Debug, Parser)]
55#[command(
56    name = "tvscreener-tui",
57    about = "TradingView screener Ratatui TUI",
58    version
59)]
60struct Cli {
61    /// Screener asset class.
62    asset: AssetArg,
63    /// Preset name (see `tvscreener presets`).
64    #[arg(long)]
65    preset: Option<String>,
66    /// Result window start (inclusive).
67    #[arg(long, default_value_t = 0)]
68    from: u32,
69    /// Max rows.
70    #[arg(long, default_value_t = 25)]
71    limit: u32,
72    /// Name/description search text.
73    #[arg(long)]
74    search: Option<String>,
75    /// Stock markets CSV (const or wire), e.g. `AMERICA`.
76    #[arg(long)]
77    markets: Option<String>,
78    /// Stock index CSV (const or wire), e.g. `SP500`.
79    #[arg(long)]
80    index: Option<String>,
81    /// Start with auto-refresh enabled (still floored to 10s).
82    #[arg(long)]
83    watch: bool,
84    /// Watch interval in seconds (default 30; minimum 10).
85    #[arg(long, default_value_t = DEFAULT_WATCH_INTERVAL_SECS)]
86    interval: f64,
87    /// Force screener debug logs (also: `TVSCREENER_DEBUG=1`).
88    #[arg(long)]
89    debug: bool,
90}

92#[tokio::main]
93async fn main() -> Result<()> {
94    let cli = Cli::parse();
95    tvscreener::logging::init_logging();
96
97    let config = ScanConfig {
98        asset: cli.asset.asset(),
99        preset: cli.preset.clone(),
100        from: cli.from,
101        limit: cli.limit,
102        search: cli.search.clone(),
103        markets: cli.markets.clone(),
104        index: cli.index.clone(),
105        filters: Vec::new(),
106    };
107
108    let interval = cli.interval.max(MIN_TUI_REFRESH_SECS);
109
110    let fields = config.resolve_fields().context("resolve fields")?;
111    let (fields, rows) = match run_scan(&config, cli.debug).await {
112        Ok(rows) => (fields, rows),
113        Err(err) => {
114            eprintln!("initial scan failed: {err:#}");
115            (fields, Vec::new())
116        }
117    };
118
119    let mut model = AppModel::new(config, fields, rows, cli.watch, interval);
120    if model.rows.is_empty() && model.error.is_none() {
121        model.status = format!("{}  (no rows)", model.status);
122    }
123
124    install_signal_handlers();
125    install_panic_hook();
126    let use_alt = !inside_gnu_screen();
127    let _guard = TerminalGuard::enter(use_alt)?;
128
129    let backend = CrosstermBackend::new(stdout());
130    let mut terminal = Terminal::new(backend)?;
131    terminal.clear()?;
132
133    let result = run_loop(&mut terminal, &mut model, cli.debug).await;
134
135    let _ = terminal.clear();
136    hard_reset_tty();
137    result
138}
139
140async fn run_loop(
141    terminal: &mut Terminal<CrosstermBackend<io::Stdout>>,
142    model: &mut AppModel,
143    debug: bool,
144) -> Result<()> {
145    loop {
146        if STOP.load(Ordering::SeqCst) {
147            break;
148        }
149
150        if model.watch_due() {
151            refresh(model, debug).await;
152        }
153
154        terminal.draw(|frame| draw(frame, model))?;
155
156        if !event::poll(Duration::from_millis(200))? {
157            continue;
158        }
159        let Event::Key(key) = event::read()? else {
160            continue;
161        };
162        if key.kind != KeyEventKind::Press {
163            continue;
164        }
165
166        if is_quit_key(key.code, key.modifiers) {
167            break;
168        }
169
170        if model.builder.input_mode != InputMode::None && handle_input_key(model, key.code) {
171            continue;
172        }
173
174        handle_global_key(model, key.code, debug).await?;
175    }
176    Ok(())
177}

179async fn handle_global_key(model: &mut AppModel, code: KeyCode, debug: bool) -> Result<()> {
180    match code {
181        KeyCode::Tab => model.cycle_view(),
182        KeyCode::Char('h') => model.toggle_help(),
183        KeyCode::Char('a') => model.toggle_watch(),
184        KeyCode::Char('c') => copy_active_pane(model),
185        KeyCode::Char('r') => {
186            if model.can_refresh() {
187                refresh(model, debug).await;
188            } else {
189                let wait = model.cooldown_secs_remaining();
190                model.status = format!("cooldown {wait}s (min {MIN_TUI_REFRESH_SECS}s)");
191            }
192        }
193        KeyCode::Char('1') => model.set_view_number(1),
194        KeyCode::Char('2') => model.set_view_number(2),
195        KeyCode::Char('3') => model.set_view_number(3),
196        KeyCode::Char('4') => model.set_view_number(4),
197        KeyCode::Up => model.scroll_by(-1),
198        KeyCode::Down => model.scroll_by(1),
199        KeyCode::PageUp => model.scroll_by(-10),
200        KeyCode::PageDown => model.scroll_by(10),
201        KeyCode::Enter if model.view == ViewMode::Builder && model.can_refresh() => {
202            refresh(model, debug).await;
203        }
204        KeyCode::Left
205            if model.view == ViewMode::Builder && model.builder.input_mode == InputMode::None =>
206        {
207            model.step_preset(-1);
208        }
209        KeyCode::Right
210            if model.view == ViewMode::Builder && model.builder.input_mode == InputMode::None =>
211        {
212            model.step_preset(1);
213        }
214        KeyCode::Char('+' | '=') if model.view == ViewMode::Builder => model.step_limit(1),
215        KeyCode::Char('-') if model.view == ViewMode::Builder => model.step_limit(-1),
216        KeyCode::Char('s') if model.view == ViewMode::Builder => start_search_input(model),
217        KeyCode::Char('f') if model.view == ViewMode::Builder => start_filter_input(model),
218        KeyCode::Char('d') if model.view == ViewMode::Builder => remove_selected_filter(model),
219        KeyCode::Char('o')
220            if model.view == ViewMode::Builder
221                && model.builder.input_mode == InputMode::FilterOp =>
222        {
223            cycle_filter_op(model, 1);
224        }
225        KeyCode::Char('j') if model.view == ViewMode::Builder => handle_builder_j(model),
226        KeyCode::Char('k') if model.view == ViewMode::Builder => handle_builder_k(model),
227        _ => {}
228    }
229    Ok(())
230}
231
232fn handle_builder_j(model: &mut AppModel) {
233    match model.builder.input_mode {
234        InputMode::FilterField => step_field_pick(model, 1),
235        InputMode::FilterOp => cycle_filter_op(model, 1),
236        InputMode::None if model.builder.focus == BuilderFocus::Filters => {
237            step_filter_select(model, 1);
238        }
239        InputMode::None => model.builder_focus_by(1),
240        InputMode::Search | InputMode::FilterValue => {}
241    }
242}
243
244fn handle_builder_k(model: &mut AppModel) {
245    match model.builder.input_mode {
246        InputMode::FilterField => step_field_pick(model, -1),
247        InputMode::FilterOp => cycle_filter_op(model, -1),
248        InputMode::None if model.builder.focus == BuilderFocus::Filters => {
249            step_filter_select(model, -1);
250        }
251        InputMode::None => model.builder_focus_by(-1),
252        InputMode::Search | InputMode::FilterValue => {}
253    }
254}
255
256/// Returns true when the key was consumed by input mode handling.
257fn handle_input_key(model: &mut AppModel, code: KeyCode) -> bool {
258    match code {
259        KeyCode::Esc => {
260            model.builder.cancel_input();
261            true
262        }
263        KeyCode::Backspace => {
264            model.builder.input_buf.pop();
265            if model.builder.input_mode == InputMode::FilterField {
266                model.builder.refresh_field_matches(model.config.asset);
267            }
268            true
269        }
270        KeyCode::Enter => {
271            commit_input(model);
272            true
273        }
274        KeyCode::Char(c) if !c.is_control() => {
275            if model.builder.input_mode == InputMode::FilterOp {
276                if c == 'o' {
277                    cycle_filter_op(model, 1);
278                }
279            } else {
280                model.builder.input_buf.push(c);
281                if model.builder.input_mode == InputMode::FilterField {
282                    model.builder.refresh_field_matches(model.config.asset);
283                }
284            }
285            true
286        }
287        _ => false,
288    }
289}

291fn start_search_input(model: &mut AppModel) {
292    model.builder.cancel_input();
293    model.builder.input_mode = InputMode::Search;
294    model.builder.input_buf = model.config.search.clone().unwrap_or_default();
295}
296
297fn start_filter_input(model: &mut AppModel) {
298    model.builder.cancel_input();
299    model.builder.input_mode = InputMode::FilterField;
300    model.builder.refresh_field_matches(model.config.asset);
301}
302
303fn step_field_pick(model: &mut AppModel, delta: isize) {
304    let len = model.builder.field_matches.len();
305    if len == 0 {
306        return;
307    }
308    let idx = model.builder.field_pick;
309    model.builder.field_pick = if delta >= 0 {
310        (idx + 1).min(len - 1)
311    } else {
312        idx.saturating_sub(1)
313    };
314}
315
316fn step_filter_select(model: &mut AppModel, delta: isize) {
317    let len = model.config.filters.len();
318    if len == 0 {
319        return;
320    }
321    let idx = model.builder.filter_selected;
322    model.builder.filter_selected = if delta >= 0 {
323        (idx + 1).min(len - 1)
324    } else {
325        idx.saturating_sub(1)
326    };
327}
328
329const fn cycle_filter_op(model: &mut AppModel, delta: isize) {
330    let len = FILTER_OPS.len();
331    let idx = model.builder.op_index;
332    model.builder.op_index = if delta >= 0 {
333        (idx + 1) % len
334    } else {
335        (idx + len - 1) % len
336    };
337}
338
339fn commit_input(model: &mut AppModel) {
340    match model.builder.input_mode {
341        InputMode::Search => {
342            let text = model.builder.input_buf.trim().to_string();
343            model.config.search = if text.is_empty() { None } else { Some(text) };
344            model.builder.cancel_input();
345        }
346        InputMode::FilterField => {
347            let Some(field) = model
348                .builder
349                .field_matches
350                .get(model.builder.field_pick)
351                .map(|f| f.field_name.clone())
352            else {
353                model.status = "pick a field (type to search, j/k, Enter)".into();
354                return;
355            };
356            model.builder.draft_field = Some(field);
357            model.builder.input_buf.clear();
358            model.builder.input_mode = InputMode::FilterOp;
359        }
360        InputMode::FilterOp => {
361            model.builder.input_mode = InputMode::FilterValue;
362            model.builder.input_buf.clear();
363        }
364        InputMode::FilterValue => {
365            let Some(left) = model.builder.draft_field.clone() else {
366                model.builder.cancel_input();
367                return;
368            };
369            let op = FILTER_OPS
370                .get(model.builder.op_index)
371                .copied()
372                .unwrap_or(FILTER_OPS[0]);
373            let value = parse_filter_value(&model.builder.input_buf);
374            model
375                .config
376                .filters
377                .push(FieldCondition::new(left, op, value));
378            model.builder.filter_selected = model.config.filters.len().saturating_sub(1);
379            model.builder.cancel_input();
380            model.status = format!("filter added ({} total)", model.config.filters.len());
381        }
382        InputMode::None => {}
383    }
384}
385
386fn parse_filter_value(raw: &str) -> Value {
387    let trimmed = raw.trim();
388    if trimmed.is_empty() {
389        return Value::Null;
390    }
391    if let Ok(n) = trimmed.parse::<f64>() {
392        return json!(n);
393    }
394    json!(trimmed)
395}
396
397fn remove_selected_filter(model: &mut AppModel) {
398    let idx = model.builder.filter_selected;
399    if idx < model.config.filters.len() {
400        model.config.filters.remove(idx);
401        if model.builder.filter_selected > 0
402            && model.builder.filter_selected >= model.config.filters.len()
403        {
404            model.builder.filter_selected = model.config.filters.len().saturating_sub(1);
405        }
406    }
407}

409async fn refresh(model: &mut AppModel, debug: bool) {
410    model.status = "Scanning...".into();
411    model.error = None;
412    match run_scan(&model.config, debug).await {
413        Ok(rows) => match model.config.resolve_fields() {
414            Ok(fields) => model.set_rows(fields, rows),
415            Err(err) => model.set_error(err.to_string()),
416        },
417        Err(err) => model.set_error(err.to_string()),
418    }
419}
420
421fn copy_active_pane(model: &mut AppModel) {
422    let payload = match model.view {
423        ViewMode::Results => serde_json::to_string_pretty(&model.rows).map_err(|e| e.to_string()),
424        ViewMode::Payload => payload_pretty(&model.config, false).map_err(|e| e.to_string()),
425        ViewMode::Codegen => Ok(render_codegen(&model.config)),
426        ViewMode::Builder | ViewMode::Help => {
427            model.status = "copy on Results, Payload, or Codegen".into();
428            return;
429        }
430    };
431    match payload {
432        Ok(text) => match copy_via_osc52(&text) {
433            Ok(()) => model.status = "copied to clipboard (OSC 52)".into(),
434            Err(err) => model.set_error(format!("clipboard write failed: {err}")),
435        },
436        Err(err) => model.set_error(format!("encode failed: {err}")),
437    }
438}

440async fn run_scan(config: &ScanConfig, debug: bool) -> Result<Vec<ScreenerRow>> {
441    Ok(tvscreener::core::get_for_asset(config.asset, |inner| {
442        configure_screener(inner, config, debug)
443    })
444    .await?)
