15/// Default auto-refresh interval when watch mode is on (seconds).
16pub const DEFAULT_WATCH_INTERVAL_SECS: f64 = 30.0;
17
18/// Minimum gap between TUI scans (manual `r` or watch).
19///
20/// Stricter than library [`crate::util::MIN_STREAM_INTERVAL_SECS`] (1s), which is only an
21/// API floor for `stream()`, not a UI default.
22pub const MIN_TUI_REFRESH_SECS: f64 = 10.0;
23
24/// Which pane the TUI is showing.
25#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
26pub enum ViewMode {
27    /// Beautified results table.
28    #[default]
29    Results,
30    /// Query builder (preset, limit, search, filters).
31    Builder,
32    /// Pretty-printed `build_payload()` JSON.
33    Payload,
34    /// Rust + CLI codegen snippets.
35    Codegen,
36    /// Key binding help.
37    Help,
38}
39
40impl ViewMode {
41    /// Cycles to the next primary pane (skips Help).
42    #[must_use]
43    pub const fn next(self) -> Self {
44        match self {
45            Self::Results => Self::Builder,
46            Self::Builder => Self::Payload,
47            Self::Payload => Self::Codegen,
48            Self::Codegen | Self::Help => Self::Results,
49        }
50    }
51
52    /// Short label for the title bar.
53    #[must_use]
54    pub const fn tab_label(self) -> &'static str {
55        match self {
56            Self::Results => "1 Results",
57            Self::Builder => "2 Builder",
58            Self::Payload => "3 Payload",
59            Self::Codegen => "4 Codegen",
60            Self::Help => "Help",
61        }
62    }
63}

79/// Text input sub-mode inside the builder pane.
80#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
81pub enum InputMode {
82    /// No active input.
83    #[default]
84    None,
85    /// Editing search text.
86    Search,
87    /// Picking a field from catalog search hits.
88    FilterField,
89    /// Cycling filter operator before value entry.
90    FilterOp,
91    /// Typing filter right-hand value.
92    FilterValue,
93}

95/// Operators offered in the simple filter wizard.
96pub const FILTER_OPS: [FilterOperator; 6] = [
97    FilterOperator::Above,
98    FilterOperator::Below,
99    FilterOperator::AboveOrEqual,
100    FilterOperator::BelowOrEqual,
101    FilterOperator::Equal,
102    FilterOperator::Match,

105/// Builder pane UI state (not sent to the API until refresh).
106#[derive(Debug, Clone)]
107pub struct BuilderUi {
108    /// Sorted preset names for the current asset (includes `(defaults)`).
109    pub preset_options: Vec<String>,
110    /// Index into [`Self::preset_options`].
111    pub preset_index: usize,
112    /// Highlighted builder row.
113    pub focus: BuilderFocus,
114    /// Selected filter row (`d` removes this one).
115    pub filter_selected: usize,
116    /// Active text input, if any.
117    pub input_mode: InputMode,
118    /// Buffer for search / field query / filter value.
119    pub input_buf: String,
120    /// Field catalog hits while adding a filter.
121    pub field_matches: Vec<FieldDef>,
122    /// Selected index in [`Self::field_matches`].
123    pub field_pick: usize,
124    /// Index into [`FILTER_OPS`] while adding a filter.
125    pub op_index: usize,
126    /// Draft field name between field pick and value entry.
127    pub draft_field: Option<String>,
128}
129
130impl BuilderUi {
131    /// Builds picker state from scan config.
132    #[must_use]
133    pub fn from_config(config: &ScanConfig) -> Self {
134        let preset_options = preset_options(config.asset);
135        let preset_index = preset_index(config.asset, config.preset.as_deref());
136        Self {
137            preset_options,
138            preset_index,
139            focus: BuilderFocus::default(),
140            filter_selected: 0,
141            input_mode: InputMode::default(),
142            input_buf: String::new(),
143            field_matches: Vec::new(),
144            field_pick: 0,
145            op_index: 0,
146            draft_field: None,
147        }
148    }
149
150    /// Refreshes preset list after asset changes (asset is fixed at launch).
151    pub fn sync_preset_index(&mut self, config: &ScanConfig) {
152        self.preset_options = preset_options(config.asset);
153        self.preset_index = preset_index(config.asset, config.preset.as_deref());
154    }
155
156    /// Updates field matches from the current query buffer.
157    pub fn refresh_field_matches(&mut self, asset: Asset) {
158        self.field_matches = search_fields(asset, &self.input_buf);
159        if self.field_pick >= self.field_matches.len() && !self.field_matches.is_empty() {
160            self.field_pick = self.field_matches.len() - 1;
161        }
162    }
163
164    /// Clears filter-add wizard state.
165    pub fn cancel_input(&mut self) {
166        self.input_mode = InputMode::None;
167        self.input_buf.clear();
168        self.field_matches.clear();
169        self.field_pick = 0;
170        self.op_index = 0;
171        self.draft_field = None;
172    }
173}

175/// CLI / env scan parameters for the TUI.
176#[derive(Debug, Clone)]
177pub struct ScanConfig {
178    /// Screener asset class.
179    pub asset: Asset,
180    /// Optional field preset name.
181    pub preset: Option<String>,
182    /// Result window start.
183    pub from: u32,
184    /// Max rows.
185    pub limit: u32,
186    /// Optional name search.
187    pub search: Option<String>,
188    /// Stock markets CSV (const or wire).
189    pub markets: Option<String>,
190    /// Stock index CSV (const or wire).
191    pub index: Option<String>,
192    /// Field conditions applied via `where_condition`.
193    pub filters: Vec<FieldCondition>,
194}
195
196impl ScanConfig {
197    /// Resolves selected fields (preset or asset defaults).
198    ///
199    /// # Errors
200    ///
201    /// Returns catalog errors from [`get_preset`].
202    pub fn resolve_fields(&self) -> crate::Result<Vec<FieldDef>> {
203        if let Some(name) = self.preset.as_deref() {
204            return get_preset(name);
205        }
206        Ok(default_fields(self.asset))
207    }
208}

210/// Pane model: config, last rows, status line, active view.
211#[derive(Debug, Clone)]
212pub struct AppModel {
213    /// Scan parameters.
214    pub config: ScanConfig,
215    /// Builder pane state.
216    pub builder: BuilderUi,
217    /// Fields used for the last scan / table columns.
218    pub fields: Vec<FieldDef>,
219    /// Last scan rows.
220    pub rows: Vec<ScreenerRow>,
221    /// Status / branding line (non-error).
222    pub status: String,
223    /// Last error message, if any.
224    pub error: Option<String>,
225    /// Active view.
226    pub view: ViewMode,
227    /// Vertical scroll offset (results rows or text panes).
228    pub scroll: usize,
229    /// Refresh counter.
230    pub ticks: u64,
231    /// Max data columns in the results table.
232    pub max_columns: usize,
233    /// When true, auto-refresh on [`Self::watch_interval`].
234    pub watch: bool,
235    /// Auto-refresh period (already floored to [`MIN_TUI_REFRESH_SECS`]).
236    pub watch_interval: Duration,
237    /// Instant of the last completed scan attempt (success or failure).
238    pub last_scan_at: Instant,
239}
240
241impl AppModel {
242    /// Builds a model after an initial scan (or empty rows on failure).
243    #[must_use]
244    pub fn new(
245        config: ScanConfig,
246        fields: Vec<FieldDef>,
247        rows: Vec<ScreenerRow>,
248        watch: bool,
249        watch_interval_secs: f64,
250    ) -> Self {
251        let watch_interval = Duration::from_secs_f64(watch_interval_secs.max(MIN_TUI_REFRESH_SECS));
252        let builder = BuilderUi::from_config(&config);
253        let status = format!(
254            "{}  rows={}  {}",
255            config.asset.as_str(),
256            rows.len(),
257            refresh_mode_label(watch, watch_interval),
258        );
259        Self {
260            builder,
261            config,
262            fields,
263            rows,
264            status,
265            error: None,
266            view: ViewMode::Results,
267            scroll: 0,
268            ticks: 0,
269            max_columns: DEFAULT_TABLE_MAX_COLUMNS,
270            watch,
271            watch_interval,
272            last_scan_at: Instant::now(),
273        }
274    }
275
276    /// Records a successful scan.
277    pub fn set_rows(&mut self, fields: Vec<FieldDef>, rows: Vec<ScreenerRow>) {
278        self.fields = fields;
279        self.rows = rows;
280        self.error = None;
281        self.scroll = 0;
282        self.ticks = self.ticks.saturating_add(1);
283        self.last_scan_at = Instant::now();
284        self.status = format!(
285            "{}  rows={}  tick={}  {}",
286            self.config.asset.as_str(),
287            self.rows.len(),
288            self.ticks,
289            refresh_mode_label(self.watch, self.watch_interval),
290        );
291    }
292
293    /// Records a scan failure (keeps previous rows).
294    pub fn set_error(&mut self, message: impl Into<String>) {
295        self.error = Some(message.into());
296        self.ticks = self.ticks.saturating_add(1);
297        self.last_scan_at = Instant::now();
298    }
299
300    /// Toggles Results ↔ Help.
301    pub const fn toggle_help(&mut self) {
302        self.view = match self.view {
303            ViewMode::Help => ViewMode::Results,
304            _ => ViewMode::Help,
305        };
306        self.scroll = 0;
307    }
308
309    /// Cycles Results → Builder → Payload → Codegen.
310    pub fn cycle_view(&mut self) {
311        if self.view == ViewMode::Help {
312            self.view = ViewMode::Results;
313        } else {
314            self.view = self.view.next();
315        }
316        self.scroll = 0;
317    }
318
319    /// Jumps to a numbered pane (`1`–`4`).
320    pub const fn set_view_number(&mut self, n: u8) {
321        self.view = match n {
322            1 => ViewMode::Results,
323            2 => ViewMode::Builder,
324            3 => ViewMode::Payload,
325            4 => ViewMode::Codegen,
326            _ => return,
327        };
328        self.scroll = 0;
329    }
330
331    /// Toggles opt-in auto-refresh (off by default).
332    pub fn toggle_watch(&mut self) {
333        self.watch = !self.watch;
334        self.status = format!(
335            "{}  rows={}  {}",
336            self.config.asset.as_str(),
337            self.rows.len(),
338            refresh_mode_label(self.watch, self.watch_interval),
339        );
340    }
341
342    /// Seconds remaining before another scan is allowed (`0` when ready).
343    #[must_use]
344    pub fn cooldown_secs_remaining(&self) -> u64 {
345        let min_gap = Duration::from_secs_f64(MIN_TUI_REFRESH_SECS);
346        let elapsed = self.last_scan_at.elapsed();
347        min_gap.saturating_sub(elapsed).as_secs()
348    }
349
350    /// Returns true when a manual or watch refresh may run (honors TUI floor).
351    #[must_use]
352    pub fn can_refresh(&self) -> bool {
353        self.cooldown_secs_remaining() == 0
354    }
355
356    /// Returns true when watch mode is on and the watch interval has elapsed.
357    #[must_use]
358    pub fn watch_due(&self) -> bool {
359        self.watch && self.last_scan_at.elapsed() >= self.watch_interval
360    }
361
362    /// Scrolls the active pane by `delta` lines.
363    pub fn scroll_by(&mut self, delta: isize) {
364        if delta >= 0 {
365            self.scroll = self
366                .scroll
367                .saturating_add(usize::try_from(delta).unwrap_or(0));
368        } else {
369            let step = usize::try_from(delta.wrapping_neg()).unwrap_or(0);
370            self.scroll = self.scroll.saturating_sub(step);
371        }
372    }
373
374    /// Moves builder focus up/down.
375    pub fn builder_focus_by(&mut self, delta: isize) {
376        let rows = [
377            BuilderFocus::Preset,
378            BuilderFocus::Limit,
379            BuilderFocus::Search,
380            BuilderFocus::Filters,
381        ];
382        let idx = rows
383            .iter()
384            .position(|&f| f == self.builder.focus)
385            .unwrap_or(0);
386        let next = if delta >= 0 {
387            (idx + 1).min(rows.len() - 1)
388        } else {
389            idx.saturating_sub(1)
390        };
391        self.builder.focus = rows[next];
392    }
393
394    /// Steps the preset picker and updates config.
395    pub fn step_preset(&mut self, delta: isize) {
396        let len = self.builder.preset_options.len();
397        if len == 0 {
398            return;
399        }
400        let idx = self.builder.preset_index;
401        let next = if delta >= 0 {
402            (idx + 1).min(len - 1)
403        } else {
404            idx.saturating_sub(1)
405        };
406        self.builder.preset_index = next;
407        super::scan::apply_preset_index(&mut self.config, next);
408        self.builder.sync_preset_index(&self.config);
409    }
410
411    /// Adjusts row limit (minimum 1, maximum 500).
412    pub fn step_limit(&mut self, delta: i32) {
413        let next = (i64::from(self.config.limit) + i64::from(delta)).clamp(1, 500);
414        self.config.limit = u32::try_from(next).unwrap_or(1);
415    }
416}
417
418fn refresh_mode_label(watch: bool, interval: Duration) -> String {
419    if watch {
420        format!("watch {}s", interval.as_secs().max(1))
421    } else {
422        "manual".into()
423    }
424}

426#[cfg(test)]
427mod tests {
428    use super::*;
429
430    fn empty_model(watch: bool, interval: f64) -> AppModel {
431        AppModel::new(
432            ScanConfig {
433                asset: Asset::Crypto,
434                preset: None,
435                from: 0,
436                limit: 5,
437                search: None,
438                markets: None,
439                index: None,
440                filters: Vec::new(),
441            },
442            Vec::new(),
443            Vec::new(),
444            watch,
445            interval,
446        )
447    }
448
449    #[test]
450    fn watch_off_by_default_and_interval_floored() {
451        let model = empty_model(false, 1.0);
452        assert!(!model.watch);
453        assert!(!model.watch_due());
454        assert_eq!(
455            model.watch_interval,
456            Duration::from_secs_f64(MIN_TUI_REFRESH_SECS)
457        );
458        assert!(!model.can_refresh());
459    }
460
461    #[test]
462    fn toggle_watch_updates_label() {
463        let mut model = empty_model(false, DEFAULT_WATCH_INTERVAL_SECS);
464        model.toggle_watch();
465        assert!(model.watch);
466        assert!(model.status.contains("watch"));
467        model.toggle_watch();
468        assert!(!model.watch);
469        assert!(model.status.contains("manual"));
470    }
471
472    #[test]
473    fn cycle_view_order() {
474        let mut model = empty_model(false, DEFAULT_WATCH_INTERVAL_SECS);
475        assert_eq!(model.view, ViewMode::Results);
476        model.cycle_view();
477        assert_eq!(model.view, ViewMode::Builder);
478        model.cycle_view();
479        assert_eq!(model.view, ViewMode::Payload);
480        model.cycle_view();
481        assert_eq!(model.view, ViewMode::Codegen);
482    }
483
484    #[test]
485    fn step_limit_clamps() {
486        let mut model = empty_model(false, DEFAULT_WATCH_INTERVAL_SECS);
487        model.step_limit(1000);
488        assert_eq!(model.config.limit, 500);
489        model.step_limit(-1000);
490        assert_eq!(model.config.limit, 1);
491    }
