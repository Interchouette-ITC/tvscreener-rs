27    #[test]
28    fn parse_ops_and_asset() {
29        assert_eq!(
30            FilterOperator::from_wire(">="),
31            Some(FilterOperator::AboveOrEqual)
32        );
33        assert_eq!(
34            FilterOperator::from_wire("in_range"),
35            Some(FilterOperator::InRange)
36        );
37        assert_eq!(parse_asset("crypto").unwrap(), Asset::Crypto);
38        assert!(parse_asset("nope").is_err());
39    }

64    #[test]
65    fn custom_payload_preview_merges_filters() {
66        let payload = custom_query_payload_preview(&PayloadPreviewOpts {
67            asset_type: "stock",
68            fields: Some("NAME,PRICE"),
69            filters: Some(
70                r#"[{"field":"PRICE","op":">","value":50},{"field":"RSI","op":"in_range","value":[30,70]}]"#,
71            ),
72            sort_by: Some("PRICE"),
73            ascending: false,
74            limit: 10,
75            indices: None,
76            markets: None,
77        })
78        .unwrap();
79        assert!(payload["columns"].as_array().unwrap().len() >= 2);
80        assert_eq!(payload["range"], serde_json::json!([0, 10]));
81        let filters = payload["filter"].as_array().unwrap();
82        assert!(filters
83            .iter()
84            .any(|f| f["left"] == "close" || f["operation"] == "greater"));
85    }

87    #[test]
88    fn build_payload_with_index_and_markets() {
89        let text = build_payload(&BuildPayloadOpts {
90            asset_type: "stock",
91            fields: Some("NAME,PRICE"),
92            filters: None,
93            sort_by: Some("MARKET_CAPITALIZATION"),
94            indices: Some("SP500,NASDAQ_100"),
95            markets: Some("AMERICA"),
96        })
97        .unwrap();
98        assert!(text.contains("SYML:SP;SPX"));
99        assert!(text.contains("SYML:NASDAQ;NDX"));
100        assert!(text.contains("america"));
101        assert_eq!(resolve_index_wires(Some("SP;SPX")).unwrap(), vec!["SP;SPX"]);
102        assert_eq!(
103            resolve_index_wires(Some("SYML:SP;SPX")).unwrap(),
104            vec!["SP;SPX"]
105        );
106        assert_eq!(resolve_index_wires(Some("SP500")).unwrap(), vec!["SP;SPX"]);
107        assert_eq!(
108            resolve_market_wires(Some("america")).unwrap(),
109            vec!["america"]
110        );
111        assert!(resolve_index_wires(Some("NOPE")).is_err());
112    }
