16/// Formats `discover_fields` search hits as text.
17#[must_use]
18pub fn format_discover_fields(asset: Asset, search_term: &str, limit: usize) -> String {
19    let limit = limit.clamp(1, 100);
20    let hits = search_fields(asset, search_term);
21    let mut lines = Vec::new();
22    for def in hits.into_iter().take(limit) {
23        let Some((const_name, resolved)) = resolve_field(asset, &def.field_name) else {
24            continue;
25        };
26        let marker = if resolved.interval {
27            " [Technical]"
28        } else {
29            ""
30        };
31        lines.push(format!("- **{const_name}**: {}{marker}", resolved.label));
32    }
33    if lines.is_empty() {
34        format!("No fields found matching '{search_term}'")
35    } else {
36        format!(
37            "Found {} fields matching '{search_term}':\n\n{}\n",
38            lines.len(),
39            lines.join("\n")
40        )
41    }
42}

44/// Lists sample field categories for an asset (keyword buckets).
45#[must_use]
46pub fn format_field_types(asset: Asset) -> String {
47    const KEYWORDS: &[&str] = &[
48        "price",
49        "volume",
50        "moving_average",
51        "rsi",
52        "macd",
53        "bollinger",
54        "earnings",
55        "dividend",
56        "market_cap",
57        "sector",
58        "recommend",
59    ];
60    let mut out = format!("Field categories for {}:\n\n", asset.as_str());
61    for keyword in KEYWORDS {
62        let names: Vec<String> = search_fields(asset, keyword)
63            .into_iter()
64            .take(5)
65            .filter_map(|def| resolve_field(asset, &def.field_name).map(|(n, _)| n))
66            .collect();
67        if names.is_empty() {
68            continue;
69        }
70        let _ = writeln!(out, "**{keyword}**:");
71        for name in names {
72            let _ = writeln!(out, "  - {name}");
73        }
74        out.push('\n');
75    }
76    out.push_str("Use discover_fields('keyword') to find more fields in each category.\n");
77    out
78}

147/// Documents filter operators accepted by `custom_query`.
148#[must_use]
149pub fn format_list_filter_operators() -> String {
150    const OPS: &[(&str, &str)] = &[
151        (">=", "Greater than or equal"),
152        (">", "Greater than"),
153        ("<=", "Less than or equal"),
154        ("<", "Less than"),
155        ("==", "Equal to"),
156        ("!=", "Not equal to"),
157        ("match", "Text contains (for text fields like SECTOR)"),
158        (
159            "in_range",
160            "Value between [min, max] (e.g. RSI between 30-70)",
161        ),
162        ("not_in_range", "Value outside [min, max]"),
163        ("crosses", "Value crosses another"),
164        ("crosses_up", "Value crosses above another"),
165        ("crosses_down", "Value crosses below another"),
166    ];
167    let mut out = String::from("Available filter operators for custom_query:\n\n");
168    for (op, desc) in OPS {
169        let _ = writeln!(out, "- **{op}**: {desc}");
170    }
171    out.push_str(
172        "\nExample filters JSON:\n\
173         ```json\n\
174         [\n\
175           {\"field\": \"PRICE\", \"op\": \">=\", \"value\": 100},\n\
176           {\"field\": \"RSI\", \"op\": \"in_range\", \"value\": [30, 70]},\n\
177           {\"field\": \"SECTOR\", \"op\": \"match\", \"value\": \"Technology Services\"}\n\
178         ]\n\
179         ```\n",
180    );
181    out
182}

215/// Formats one preset's fields as text.
216///
217/// # Errors
218///
219/// Propagates unknown-preset errors from [`get_preset`].
220pub fn format_get_preset(name: &str) -> Result<String> {
221    let fields = get_preset(name)?;
222    let mut out = format!("Preset `{name}` ({} fields):\n\n", fields.len());
223    for field in fields {
224        let _ = writeln!(out, "- {} (`{}`)", field.label, field.field_name);
225    }
226    Ok(out)
227}

229/// Renders rows as a Markdown table (symbol + selected labels).
230#[must_use]
231pub fn format_rows_markdown(rows: &[ScreenerRow], max_rows: usize) -> String {
232    if rows.is_empty() {
233        return "No results found matching the criteria.".into();
234    }
235    let rows: Vec<&ScreenerRow> = rows.iter().take(max_rows).collect();
236    let mut labels: Vec<String> = rows
237        .iter()
238        .flat_map(|r| r.data.keys().cloned())
239        .collect::<BTreeSet<_>>()
240        .into_iter()
241        .collect();
242    // Prefer a stable short set when many columns
243    if labels.len() > 12 {
244        labels.truncate(12);
245    }
246
247    let mut header = String::from("| Symbol |");
248    let mut sep = String::from("| --- |");
249    for label in &labels {
250        let _ = write!(header, " {label} |");
251        sep.push_str(" --- |");
252    }
253    header.push('\n');
254    sep.push('\n');
255
256    let mut body = String::new();
257    for row in rows {
258        let _ = write!(body, "| {} |", row.symbol);
259        for label in &labels {
260            let cell = row
261                .data
262                .get(label)
263                .map_or_else(|| "--".into(), format_value);
264            let _ = write!(body, " {cell} |");
265        }
266        body.push('\n');
267    }
268    format!("{header}{sep}{body}")
269}
