15/// Preset names for an asset (`stock_*`, `crypto_*`, …) plus a defaults entry.
16#[must_use]
17pub fn preset_options(asset: Asset) -> Vec<String> {
18    let prefix = format!("{}_", asset.as_str());
19    let mut names: Vec<String> = list_presets()
20        .into_iter()
21        .filter(|name| name.starts_with(&prefix))
22        .collect();
23    names.sort();
24    names.insert(0, "(defaults)".into());
25    names
26}

28/// Index of `preset` in [`preset_options`] (`0` when unset or unknown).
29#[must_use]
30pub fn preset_index(asset: Asset, preset: Option<&str>) -> usize {
31    let options = preset_options(asset);
32    preset.map_or(0, |name| {
33        options.iter().position(|p| p == name).unwrap_or(0)
34    })
35}

37/// Sets `config.preset` from a preset picker index.
38pub fn apply_preset_index(config: &mut ScanConfig, index: usize) {
39    let options = preset_options(config.asset);
40    let picked = options.get(index).map(String::as_str);
41    config.preset = match picked {
42        None | Some("(defaults)") => None,
43        Some(name) => Some(name.to_string()),
44    };
45}

47/// Configures a screener from TUI scan parameters.
48///
49/// # Errors
50///
51/// Propagates field resolution, filter, search, and stock market errors.
52pub fn configure_screener(inner: &mut Screener, config: &ScanConfig, debug: bool) -> Result<()> {
53    let fields = config.resolve_fields()?;
54    if debug || crate::logging::env_debug_enabled() {
55        inner.set_debug(true);
56    }
57    inner
58        .select(fields)
59        .set_range(config.from, config.from.saturating_add(config.limit));
60    if let Some(q) = config.search.as_deref().filter(|s| !s.is_empty()) {
61        inner.search(q)?;
62    }
63    for condition in &config.filters {
64        inner.where_condition(condition.clone())?;
65    }
66    if matches!(config.asset, Asset::Stock) {
67        apply_stock_index_markets(inner, config.index.as_deref(), config.markets.as_deref())?;
68    } else if config.markets.is_some() || config.index.is_some() {
69        return Err(crate::TvscreenerError::InvalidRequest(
70            "--markets / --index only apply to stock".into(),
71        ));
72    }
73    Ok(())
74}

76/// Builds the scanner POST JSON for the current config (no network).
77///
78/// # Errors
79///
80/// Propagates [`configure_screener`] and [`Screener::build_payload`] errors.
81pub fn build_payload(config: &ScanConfig, debug: bool) -> Result<Value> {
82    with_asset_screener(config.asset, |inner| {
83        configure_screener(inner, config, debug)?;
84        inner.build_payload()
85    })
86}

88/// Pretty JSON for the payload pane.
89///
90/// # Errors
91///
92/// Propagates [`build_payload`] or JSON serialization errors.
93pub fn payload_pretty(config: &ScanConfig, debug: bool) -> Result<String> {
94    let value = build_payload(config, debug)?;
95    serde_json::to_string_pretty(&value).map_err(Into::into)
96}

98/// Rust + CLI snippets for the codegen pane.
99#[must_use]
100pub fn render_codegen(config: &ScanConfig) -> String {
101    let mut out = String::new();
102    out.push_str("// Rust\n");
103    out.push_str(&render_rust_snippet(config));
104    out.push_str("\n\n# CLI\n");
105    out.push_str(&render_cli_command(config));
106    out
107}

109const fn typed_screener_name(asset: Asset) -> &'static str {
110    match asset {
111        Asset::Stock => "StockScreener",
112        Asset::Crypto => "CryptoScreener",
113        Asset::Forex => "ForexScreener",
114        Asset::Bond => "BondScreener",
115        Asset::Futures => "FuturesScreener",
116        Asset::Coin => "CoinScreener",
117    }
118}

120fn render_rust_snippet(config: &ScanConfig) -> String {
121    let screener = typed_screener_name(config.asset);
122    let mut lines = vec![
123        "use serde_json::json;".to_string(),
124        format!("use tvscreener::core::{screener};"),
125        "use tvscreener::filter::{FieldCondition, FilterOperator};".to_string(),
126    ];
127    if config.preset.is_some() {
128        lines.push("use tvscreener::field::get_preset;".to_string());
129    }
130    if matches!(config.asset, Asset::Stock) && (config.index.is_some() || config.markets.is_some())
131    {
132        lines.push("use tvscreener::resolve::apply_stock_index_markets;".to_string());
133    }
134    lines.push(String::new());
135    lines.push(format!("let mut screener = {screener}::new();"));
136    if let Some(preset) = config.preset.as_deref() {
137        lines.push(format!("let fields = get_preset({preset:?})?;"));
138        lines.push("screener.select(fields);".to_string());
139    }
140    lines.push(format!(
141        "screener.set_range({}, {});",
142        config.from,
143        config.from.saturating_add(config.limit)
144    ));
145    if let Some(q) = config.search.as_deref().filter(|s| !s.is_empty()) {
146        lines.push(format!("screener.search({q:?})?;"));
147    }
148    for f in &config.filters {
149        lines.push(format!(
150            "screener.where_condition(FieldCondition::new({:?}, FilterOperator::{:?}, json!({})))?;",
151            f.left,
152            f.operation,
153            filter_value_literal(&f.value)
154        ));
155    }
156    if matches!(config.asset, Asset::Stock) {
157        if let Some(index) = config.index.as_deref() {
158            lines.push(format!(
159                "apply_stock_index_markets(screener.inner_mut(), Some({index:?}), None)?;"
160            ));
161        }
162        if let Some(markets) = config.markets.as_deref() {
163            let index_arg = config
164                .index
165                .as_deref()
166                .map_or_else(|| "None".into(), |i| format!("Some({i:?})"));
167            lines.push(format!(
168                "apply_stock_index_markets(screener.inner_mut(), {index_arg}, Some({markets:?}))?;"
169            ));
170        }
171    }
172    lines.push("let rows = screener.get().await?;".to_string());
173    lines.join("\n")
174}

176fn filter_value_literal(value: &Value) -> String {
177    match value {
178        Value::String(s) => format!("{s:?}"),
179        other => other.to_string(),
180    }
181}

183fn render_cli_command(config: &ScanConfig) -> String {
184    let mut parts = vec![
185        "tvscreener-tui".to_string(),
186        config.asset.as_str().to_string(),
187    ];
188    if let Some(preset) = &config.preset {
189        parts.push(format!("--preset {preset}"));
190    }
191    if config.from != 0 {
192        parts.push(format!("--from {}", config.from));
193    }
194    if config.limit != 25 {
195        parts.push(format!("--limit {}", config.limit));
196    }
197    if let Some(q) = config.search.as_deref().filter(|s| !s.is_empty()) {
198        parts.push(format!("--search {q:?}"));
199    }
200    if let Some(markets) = &config.markets {
201        parts.push(format!("--markets {markets}"));
202    }
203    if let Some(index) = &config.index {
204        parts.push(format!("--index {index}"));
205    }
206    // Field filters are not exposed on the CLI yet; note equivalent scan command.
207    let mut scan = parts.clone();
208    scan[0] = "tvscreener".into();
209    scan.insert(1, "scan".into());
210    if config.filters.is_empty() {
211        parts.join(" ")
212    } else {
213        format!(
214            "{}\n# filters: use Rust API or TUI builder (CLI filter flags not yet available)\n{}",
215            parts.join(" "),
216            scan.join(" ")
217        )
218    }
219}

228    fn sample_config() -> ScanConfig {
229        ScanConfig {
230            asset: Asset::Crypto,
231            preset: Some("crypto_price".into()),
232            from: 0,
233            limit: 10,
234            search: Some("BTC".into()),

245    #[test]
246    fn preset_options_start_with_defaults() {
247        let opts = preset_options(Asset::Crypto);
248        assert_eq!(opts[0], "(defaults)");
249        assert!(opts.iter().any(|n| n == "crypto_price"));
250    }
251
252    #[test]
253    fn build_payload_includes_filter() {
254        let config = sample_config();
255        let payload = build_payload(&config, false).expect("payload");
256        let filters = payload["filter"].as_array().expect("filter array");
257        assert!(filters.iter().any(|f| f["left"] == "close"));
258    }
259
260    #[test]
261    fn codegen_mentions_typed_screener_and_cli() {
262        let text = render_codegen(&sample_config());
263        assert!(text.contains("CryptoScreener"));
264        assert!(text.contains("tvscreener-tui"));
265        assert!(text.contains("where_condition"));
266    }
267
268    #[test]
269    fn apply_preset_index_clears_on_defaults() {
270        let mut config = sample_config();
271        apply_preset_index(&mut config, 0);
272        assert!(config.preset.is_none());
273    }
274
275    #[test]
276    fn resolve_fields_uses_preset() {
277        let fields = sample_config().resolve_fields().expect("fields");
278        assert!(!fields.is_empty());
279        let preset_only = get_preset("crypto_price").expect("preset");
280        assert_eq!(fields.len(), preset_only.len());
281        assert_ne!(fields, default_fields(Asset::Crypto));
282    }
